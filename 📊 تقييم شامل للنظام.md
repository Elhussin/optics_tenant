class FlexiblePrice(BaseModel):
    # ... الحقول الحالية ...
    
    # ربط بالشركاء
    partner = models.ForeignKey(
        'crm.Partner', on_delete=models.SET_NULL, 
        null=True, blank=True,
        help_text="سعر خاص لشريك معين (تأمين/تقسيط/جملة)"
    )
    
    # نوع السعر
    price_type = models.CharField(max_length=20, choices=[
        ('retail', 'تجزئة'),
        ('wholesale', 'جملة'),
        ('insurance', 'تأمين'),
        ('corporate', 'شركات'),
    ], default='retail')
    
    # خصم إضافي
    discount_percentage = models.DecimalField(max_digits=5, decimal_places=2, default=0)
    
    @classmethod
    def get_price_for_order(cls, variant, customer=None, partner=None, 
                            branch=None, quantity=1, order_type='retail'):
        """
        الحصول على أفضل سعر للطلب بناءً على:
        1. العميل
        2. الشريك (تأمين/تقسيط)
        3. الفرع
        4. الكمية
        5. نوع البيع (تجزئة/جملة)
        """
        prices = cls.objects.filter(
            variant=variant,
            is_active=True
        ).order_by('-priority')
        
        for price in prices:
            if price.is_valid(customer, partner, branch, quantity, order_type):
                return price.calculate_final_price()
        
        # السعر الافتراضي
        return variant.selling_price

# apps/accounting/services/journal_service.py
def create_sale_journal_entries(invoice):
    """إنشاء قيود محاسبية من فاتورة بيع"""
    entries = []
    
    # مدين: العميل/النقد
    entries.append(JournalEntry(
        account=get_account('receivables' if invoice.remaining_amount > 0 else 'cash'),
        debit=invoice.total_amount,
        credit=Money(0, invoice.currency),
    ))
    
    # دائن: إيرادات المبيعات
    entries.append(JournalEntry(
        account=get_account('sales_revenue'),
        debit=Money(0, invoice.currency),
        credit=invoice.subtotal,
    ))
    
    # دائن: ضريبة القيمة المضافة
    if invoice.tax_amount > 0:
        entries.append(JournalEntry(
            account=get_account('vat_payable'),
            debit=Money(0, invoice.currency),
            credit=invoice.tax_amount,
        ))
    
    return entries
